from backend.brain.plan import Plan


class Planner:
    """
    Converts user input into an executable Plan.

    The planner NEVER executes anything.
    It only decides what should happen.
    """

    def create_plan(self, user_input: str) -> Plan:

        text = user_input.strip()

        lower = text.lower()

        # --------------------------------------------------
        # Calculator
        # --------------------------------------------------

        if any(op in text for op in ["+", "-", "*", "/", "%"]):

            plan = Plan(
                goal="Solve mathematical expression",
                confidence=0.98,
            )

            plan.add_step(
                action="execute",
                target="calculator",
                expression=text,
            )

            return plan

        # --------------------------------------------------
        # LLM
        # --------------------------------------------------

        plan = Plan(
            goal="Generate response",
            confidence=0.90,
        )

        plan.add_step(
            action="generate",
            target="llm",
            prompt=text,
        )

        return plan

from backend.brain.plan import Plan, StepStatus
from backend.core.logger import get_logger
from backend.llm.factory import create_llm
from backend.tools.calculator import calculate


class Executor:
    """
    Executes a Plan generated by the Planner.

    The Executor never creates plans.
    It only executes them.
    """

    def __init__(self):
        self.logger = get_logger("Executor")
        self.llm = create_llm()

    def execute(self, plan: Plan) -> str:
        """
        Execute every step in the plan.
        """

        final_result = ""

        while True:

            step = plan.next_step()

            if step is None:
                break

            step.status = StepStatus.RUNNING

            try:

                # -----------------------------------------
                # Calculator
                # -----------------------------------------

                if step.target == "calculator":

                    expression = step.parameters["expression"]

                    result = calculate(expression)

                # -----------------------------------------
                # LLM
                # -----------------------------------------

                elif step.target == "llm":

                    prompt = step.parameters["prompt"]

                    result = self.llm.generate(prompt)

                else:

                    raise ValueError(
                        f"Unknown target: {step.target}"
                    )

                step.result = result
                step.status = StepStatus.COMPLETED

                final_result = result

            except Exception as e:

                step.result = str(e)
                step.status = StepStatus.FAILED

                self.logger.exception(e)

                final_result = (
                    "Sorry, execution failed."
                )

                break

        plan.completed = plan.is_finished()

        return final_result

from collections import deque
from backend.core.logger import get_logger


class BrainState:
    """
    Short-term working memory for Brain

    This replaces the messy history list in Assistant.
    """

    def __init__(self, max_size: int = 20):
        self.logger = get_logger("BrainState")

        self.memory = deque(maxlen=max_size)

    def add(self, role: str, content: str):
        self.memory.append({
            "role": role,
            "content": content
        })

    def get_recent(self, limit: int = 10):
        return list(self.memory)[-limit:]

    def clear(self):
        self.memory.clear()

    def format_for_prompt(self):
        """
        Converts memory into LLM-friendly format
        """
        return "\n".join(
            f"{m['role']}: {m['content']}"
            for m in self.memory
        )

class Bootstrap:

    def __init__(self):
        self.logger = get_logger("Bootstrap")

        self.bus = EventBus()
        self.state = AudioState()

        self.brain = Brain(self.bus)
        self.recorder = StreamRecorder()
        self.stt = WhisperSTT()

        self.tts = PlaybackManager(
            model_path="/Users/ayushsharma/Downloads/en_US-lessac-medium.onnx"
        )

    def start(self):

        self.logger.info("MARC FULL DUPLEX MODE STARTED")

        self.brain.start()

        # RESPONSE HANDLER
        def handle_response(event: Event):

            text = event.data.get("text")
            if not text:
                return

            print(f"\nMARC: {text}")

            self.tts.speak(text, self.state)

        self.bus.subscribe("response.generated", handle_response)

        # AUDIO CALLBACK
        def on_audio(file):

            text = self.stt.transcribe(file)

            if not text:
                return

            print(f"\nUSER: {text}")

            self.bus.publish(
                Event(
                    type="text.recognized",
                    data={"text": text},
                    source="stt"
                )
            )

        # START STREAMING MIC LOOP
        self.recorder.listen_stream(on_audio)

from collections import defaultdict
from typing import Callable, Dict, List
from backend.runtime.event import Event


class EventBus:
    """
    Simple synchronous event bus (upgrade later to async).
    """

    def __init__(self):
        self.subscribers: Dict[str, List[Callable]] = defaultdict(list)

    def subscribe(self, event_type: str, handler: Callable):
        self.subscribers[event_type].append(handler)

    def unsubscribe(self, event_type: str, handler: Callable):
        if handler in self.subscribers[event_type]:
            self.subscribers[event_type].remove(handler)

    def publish(self, event: Event):
        handlers = self.subscribers.get(event.type, [])

        for handler in handlers:
            try:
                handler(event)
            except Exception as e:
                print(f"[EventBus] Handler error: {e}")

from backend.core.logger import get_logger
from backend.runtime.event_bus import EventBus


class Engine:
    """
    MARC Runtime Engine

    Responsibilities:
    - Manage lifecycle of all components
    - Run update loop
    - Provide event bus communication layer
    """

    def __init__(self):
        self.logger = get_logger("Engine")

        self.bus = EventBus()

        self.components = []
        self.running = False

    # ------------------------------------
    # Component Management
    # ------------------------------------

    def register(self, component):
        """
        Add a runtime component.
        """
        self.components.append(component)
        self.logger.info(f"Registered component: {component.name}")

    # ------------------------------------
    # Lifecycle
    # ------------------------------------

    def start(self):
        """
        Start all components.
        """
        self.logger.info("Starting MARC Engine...")

        self.running = True

        for c in self.components:
            try:
                c.start()
            except Exception as e:
                self.logger.exception(
                    f"Failed to start {c.name}: {e}"
                )

        self.logger.info("Engine started successfully.")

    def run(self):
        """
        Main loop.
        """
        self.logger.info("Engine running...")

        while self.running:

            try:
                for c in self.components:
                    c.update()

            except KeyboardInterrupt:
                self.logger.info("Shutdown requested by user.")
                self.stop()

            except Exception as e:
                self.logger.exception(f"Engine crash: {e}")

                # You can decide:
                # stop or continue
                self.stop()

    def stop(self):
        """
        Shutdown all components safely.
        """
        self.logger.info("Stopping MARC Engine...")

        self.running = False

        for c in reversed(self.components):
            try:
                c.stop()
            except Exception as e:
                self.logger.exception(
                    f"Error stopping {c.name}: {e}"
                )

        self.logger.info("Engine stopped cleanly.")

from abc import ABC, abstractmethod


class Component(ABC):
    """
    Base Component for MARC v2

    Every system module must follow this lifecycle:
    - start()
    - update()
    - stop()
    """

    def __init__(self, name: str):
        self.name = name
        self.running = False

    # ------------------------------------
    # Lifecycle Methods
    # ------------------------------------

    @abstractmethod
    def start(self):
        """
        Called once when engine starts.
        """
        self.running = True

    @abstractmethod
    def update(self):
        """
        Called repeatedly by engine loop.
        """
        pass

    @abstractmethod
    def stop(self):
        """
        Called when engine shuts down.
        """
        self.running = False

from backend.skills.calculator import CalculatorSkill
from backend.skills.llm import LLMSkill


class SkillRegistry:
    """
    Stores and executes all available skills.
    """

    def __init__(self):
        self.skills = {}
        self.register(CalculatorSkill())
        self.register(LLMSkill())

    def register(self, skill):
        self.skills[skill.name] = skill

    def get(self, name):
        return self.skills.get(name)

    def execute(self, name, text):
        skill = self.get(name)

        if skill is None:
            raise ValueError(f"Unknown skill: {name}")

        return skill.execute(text)

from abc import ABC, abstractmethod
from typing import Any


class Skill(ABC):
    """
    Base class for every MARC skill.

    Every capability (LLM, Calculator, Browser, Vision, etc.)
    inherits from this class.
    """

    name: str = ""
    description: str = ""

    @abstractmethod
    def execute(self, **kwargs) -> Any:
        """
        Execute the skill.

        Returns the result produced by the skill.
        """
        pass
