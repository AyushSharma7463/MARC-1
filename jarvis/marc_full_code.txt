    24	        """
    25	        Blocking playback.
    26	        """
    27	
    28	        if not text:

    30

    32

    36	                self.tts.speak(text)

    38	            finally:

    43	        Start playback in a background thread.
    44	        Returns immediately.
    45	        """
    46	
    47	        # Don't overlap speech
    48	        if self.is_playing():
    49	            self.stop()

    52	            target=self.speak,
    53	            args=(text,),
    54	            daemon=True,
    55	        )

    57	        self._thread.start()

    62	        """

    16	class VoiceActivityDetector:
    17	    """
    18	    Energy-based Voice Activity Detection (VAD).
    19	
    20	    States:
    21	        - Idle
    22	        - Speech Detected
    23	        - Recording
    24	        - Stop Recording
    25	    """
    26	
    27	    def __init__(
    28	        self,
    29	        threshold: float = 0.015,
    30	        speech_frames: int = 3,
    31	        silence_frames: int = 50,
    32	    ):
    33	        """
    34	        Parameters
    35	        ----------
    36	        threshold
    37	            Minimum RMS energy considered speech.
    38	
    39	        speech_frames
    40	            Number of consecutive speech chunks required
    41	            before recording starts.
    42	
    43	        silence_frames
    44	            Number of consecutive silence chunks required
    45	            before recording stops.
    46	        """
    47	
    48	        self.threshold = threshold
    49	
    50	        self.required_speech_frames = speech_frames
    51	        self.required_silence_frames = silence_frames
    52	
    53	        self.speech_count = 0
    54	        self.silence_count = 0
    55	
    56	        self.recording = False
    57	
    58	        self.logger = get_logger("VAD")
    59	
    60	    def rms(self, audio_chunk: np.ndarray) -> float:
    61	        """
    62	        Compute RMS energy.
    63	        """
    64	
    65	        if len(audio_chunk) == 0:
    66	            return 0.0
    67	
    68	        return np.sqrt(np.mean(audio_chunk ** 2))
    69	
    70	    def is_speech(self, audio_chunk: np.ndarray) -> bool:
    71	        """
    72	        Returns True if the audio chunk contains speech.
    73	        """
    74	
    75	        energy = self.rms(audio_chunk)
    76	
    77	        self.logger.info(f"Energy: {energy:.5f}")
    78	
    79	        return energy >= self.threshold
    80	
    81	    def update(self, audio_chunk: np.ndarray):
    82	        """
    83	        Process one audio chunk.
    84	
    85	        Returns
    86	        -------
    87	        idle
    88	            Waiting for speech.
    89	
    90	        start
    91	            Start recording.
    92	
    93	        recording
    94	            Continue recording.
    95	
    96	        stop
    97	            Stop recording.
    98	        """
    99	
   100	        is_speech = self.is_speech(audio_chunk)
   101	
   102	        # -----------------------------
   103	        # Waiting for speech
   104	        # -----------------------------
   105	        if not self.recording:
   106	
   107	            if is_speech:
   108	
   109	                self.speech_count += 1
   110	
   111	                if self.speech_count >= self.required_speech_frames:
   112	
   113	                    self.recording = True
   114	                    self.silence_count = 0
   115	
   116	                    self.logger.info("Speech detected")
   117	
   118	                    return "start"
   119	
   120	            else:
   121	
   122	                self.speech_count = 0
   123	
   124	            return "idle"
   125	
   126	        # -----------------------------
   127	        # Already recording
   128	        # -----------------------------
   129	        if is_speech:
   130	
   131	            self.silence_count = 0
   132	
   133	            return "recording"
   134	
   135	        self.silence_count += 1
   136	
   137	        if self.silence_count >= self.required_silence_frames:
   138	
   139	            self.recording = False
   140	
   141	            self.speech_count = 0
   142	            self.silence_count = 0
   143	
   144	            self.logger.info("Silence detected")
   145	
   146	            return "stop"
   147	
   148	        return "recording"

     8	class OllamaClient:
     9	    """
    10	    Client for communicating with the local Ollama server.
    11	    """
    12	
    13	    def __init__(self, config: LLMConfig | None = None):
    14	        self.config = config or LLMConfig()
    15	        self.logger = get_logger("LLM")
    16	
    17	    def generate(self, prompt: str) -> str:
    18	        start = time.time()
    19	
    20	        self.logger.info("Request sent to Ollama")
    21	
    22	        url = f"{self.config.host}/api/generate"
    23	
    24	        payload = {
    25	            "model": self.config.model,
    26	            "prompt": prompt,
    27	            "stream": False,
    28	            "options": {
    29	                "temperature": self.config.temperature,
    30	            },
    31	        }
    32	
    33	        try:
    34	            response = requests.post(
    35	                url,
    36	                json=payload,
    37	                timeout=120,
    38	            )
    39	
    40	            response.raise_for_status()
    41	
    42	            result = response.json()["response"]
    43	
    44	            duration = round(time.time() - start, 2)
    45	
    46	            self.logger.info(
    47	                f"Response received in {duration}s"
    48	            )
    49	
    50	            return result
    51	
    52	        except Exception as e:
    53	            self.logger.error(
    54	                f"Ollama request failed: {e}"
    55	            )
    56	            raise

   103	                original_text.lower().split()

   154

   156

   158

   162

     8	    Simple synchronous event bus (upgrade later to async).

    26	                handler(event)
    27	            except Exception as e:
    28	                print(f"[EventBus] Handler error: {e}")

     4	from threading import Lock
     5

     7	
     8	
     9	class PiperTTS:
    10	    """
    11	    Offline Text-to-Speech using Piper.
    12	
    13	    Features
    14	    --------
    15	    - Thread-safe playback
    16	    - Interruptible speech
    17	    - Prevents overlapping audio
    18	    - Ready for streaming responses
    19	    """
    20	
    21	    def __init__(
    22	        self,
    23	        model_path: str = "/Users/ayushsharma/Downloads/en_US-lessac-medium.onnx",
    24	    ):
    25	        self.logger = get_logger("PiperTTS")
    26	
    27	        self.model_path = Path(model_path)
    28	
    29	        if not self.model_path.exists():
    30	            raise FileNotFoundError(
    31	                f"Model not found: {self.model_path}"
    32	            )
    33	
    34	        # afplay process
    35	        self.player = None
    36	
    37	        # Prevent multiple playback threads
    38	        self.lock = Lock()
    39	
    40	        self.logger.info("Piper TTS initialized")
    41	
    42	    def speak(self, text: str):
    43	        """
    44	        Convert text to speech and play it.
    45	
    46	        If another speech is already playing,
    47	        stop it first.
    48	        """
    49	
    50	        if not text or not text.strip():
    51	            return
    52	
    53	        text = text.strip()
    54	
    55	        with self.lock:
    56	
    57	            # Interrupt previous playback
    58	            self.stop()
    59	
    60	            self.logger.info(f"TTS Input: {text}")
    61	
    62	            with tempfile.NamedTemporaryFile(
    63	                suffix=".wav",
    64	                delete=False,
    65	            ) as tmp:
    66	
    67	                output_wav = tmp.name
    68	
    69	            try:
    70	
    71	                # -------------------------
    72	                # Generate speech
    73	                # -------------------------
    74	                cmd = [
    75	                    "piper",
    76	                    "--model",
    77	                    str(self.model_path),
    78	                    "--output_file",
    79	                    output_wav,
    80	                ]
    81	
    82	                process = subprocess.Popen(
    83	                    cmd,
    84	                    stdin=subprocess.PIPE,
    85	                    stdout=subprocess.PIPE,
    86	                    stderr=subprocess.PIPE,
    87	                    text=True,
    88	                )
    89	
    90	                _, stderr = process.communicate(
    91	                    input=text
    92	                )
    93	
    94	                if process.returncode != 0:
    95	                    self.logger.error(
    96	                        f"Piper failed:\n{stderr}"
    97	                    )
    98	                    return
    99	
   100	                # -------------------------
   101	                # Play audio
   102	                # -------------------------
   103	                self.player = subprocess.Popen(
   104	                    ["afplay", output_wav],
   105	                    stdout=subprocess.DEVNULL,
   106	                    stderr=subprocess.DEVNULL,
   107	                )
   108	
   109	                self.logger.info(
   110	                    "Playback started"

     1	import re

     4	class WakeWordDetector:
     5	    """
     6	    Detects wake words and removes them
     7	    from the beginning of the user's speech.
     8	    """
     9	
    10	    WAKE_WORDS = (
    11	        "jarvis",
    12	        "hello jarvis",
    13	        "hi jarvis",
    14	        "ok jarvis",
    15	        "okay jarvis",
    16	        "hey jarvis",
    17	        "hey marc",
    18	        "marc",
    19	        "hey",
    20	    )
    21	
    22	    def detect(self, text: str) -> tuple[bool, str]:
    23	        """
    24	        Returns
    25	        -------
    26	        (activated, cleaned_text)
    27	        """
    28	
    29	        if not text:
    30	            return False, ""
    31	
    32	        text = self._normalize(text)
    33	
    34	        for wake_word in sorted(
    35	            self.WAKE_WORDS,
    36	            key=len,
    37	            reverse=True,
    38	        ):
    39	
    40	            if text.startswith(wake_word):
    41	
    42	                cleaned = text[len(wake_word):].strip()
    43	
    44	                return True, cleaned
    45	
    46	        return False, text
    47	
    48	    @staticmethod
    49	    def _normalize(text: str) -> str:
    50	
    51	        text = text.lower()
    52	
    53	        text = re.sub(r"[^\w\s]", "", text)
    54	
    55	        text = " ".join(text.split())
    56	
    57	        return text
