logger = get_logger("RunVoice")


def main():
    """
    Continuous Voice Assistant
    """

    logger.info("=" * 60)
    logger.info("Starting MARC Voice Assistant")
    logger.info("=" * 60)

    pipeline = VoicePipeline()

    try:

        while True:

            pipeline.run()

            print()

    except KeyboardInterrupt:
     logger.info("Stopping MARC...")
     print("\nGoodbye!")

    except Exception as e:

        logger.exception(f"Fatal Error: {e}")

    finally:

        logger.info("Voice Assistant Closed.")


if __name__ == "__main__":
    main()

from backend.audio.stream_recorder import StreamRecorder
from backend.pipeline.assistant import Assistant
from backend.stt.whisper import WhisperSTT
from backend.core.logger import get_logger
from backend.wakeword.detector import WakeWordDetector
from backend.state.conversation import ConversationManager


class VoicePipeline:
    """
    MARC Voice Pipeline

        Stream Recorder
              │
              ▼
        Whisper STT
              │
              ▼
      Wake Word Detector
              │
              ▼
    Conversation Manager
              │
              ▼
         Assistant
              │
              ▼
          Response
    """

    def __init__(self):
        self.logger = get_logger("VoicePipeline")

        self.recorder = StreamRecorder()
        self.stt = WhisperSTT()
        self.assistant = Assistant()
        self.wakeword = WakeWordDetector()
        self.conversation = ConversationManager()

    def run(self):
        """
        Run one complete voice interaction.
        """

        try:

            # ----------------------------------------
            # Listen for Speech
            # ----------------------------------------

            self.logger.info("Listening for speech...")

            audio_file = self.recorder.record()

            if audio_file is None:
                self.logger.warning("No recording captured.")
                return

            self.logger.info(f"Audio saved: {audio_file.name}")

            # ----------------------------------------
            # Speech-to-Text
            # ----------------------------------------

            self.logger.info("Transcribing audio...")

            user_text = self.stt.transcribe(audio_file)

            if not user_text or not user_text.strip():
                self.logger.warning("No speech recognized.")
                return

            print(f"\nYou : {user_text}")

            # ----------------------------------------
            # Conversation Sleeping
            # ----------------------------------------

            if not self.conversation.is_active():

                activated, cleaned_text = self.wakeword.detect(user_text)

                if not activated:

                    self.logger.info("Wake word not detected.")

                    print("\nWaiting for wake word...\n")

                    return

                self.logger.info("Wake word detected.")

                self.conversation.activate()

                # User only said the wake word

                if cleaned_text == "":

                    print("\nMARC: Yes?\n")

                    return

                user_text = cleaned_text

                print(f"\nCommand : {user_text}")

            # ----------------------------------------
            # Conversation Active
            # ----------------------------------------

            else:

                self.logger.info("Conversation already active.")

            # ----------------------------------------
            # Assistant
            # ----------------------------------------

            self.logger.info("Generating response...")

            response = self.assistant.chat(user_text)

            if response is None:
                self.logger.warning("Assistant returned no response.")
                return

            print(f"\nMARC: {response}")

            print("\n----------------------------------------")

            self.logger.info("Interaction complete.")

        except KeyboardInterrupt:

            self.logger.info("Interrupted by user.")

            raise

        except Exception as e:

            self.logger.exception(f"Voice pipeline failed: {e}")

from backend.core.logger import get_logger
from backend.llm.factory import create_llm
from backend.pipeline.intent_router import IntentRouter
from backend.tools.calculator import calculate


class Assistant:
    """
    Main Assistant Pipeline

    Responsibilities:
    - Receive user input
    - Detect intent
    - Call tools if needed
    - Build prompts
    - Talk to the LLM
    - Maintain conversation memory
    - Summarize long conversations
    """

    def __init__(self):
        self.llm = create_llm()
        self.router = IntentRouter()
        self.logger = get_logger("Assistant")

        self.history = []
        self.summary = ""

    def summarize_history(self) -> str:
        """
        Summarize older conversation history once it becomes large.
        """

        if len(self.history) < 12:
            return self.summary

        summary_input = "\n".join(
            f"{role}: {message}"
            for role, message in self.history[:-6]
        )

        prompt = f"""
Summarize the following conversation.

Keep only:
- Important facts
- User preferences
- Important context

Ignore greetings and small talk.

Conversation:

{summary_input}

Summary:
"""

        try:
            summary = self.llm.generate(prompt).strip()

            if summary:
                return summary

            return self.summary

        except Exception:
            return self.summary

    def build_prompt(self, user_input: str) -> str:
        """
        Build the prompt sent to the LLM.
        """

        new_summary = self.summarize_history()

        if new_summary:
            self.summary = new_summary

        recent_context = "\n".join(
            f"{role}: {message}"
            for role, message in self.history[-6:]
        )

        system_prompt = """
You are Jarvis.

Identity Rules:
- Your name is Jarvis.
- Never invent another identity.
- Never claim to be another person.
- Never say your name is Holden, ChatGPT, Claude, Gemini, or any other assistant.
- Stay in character.

Communication Style:
- Professional
- Friendly
- Concise
- Helpful

If the user calls you Jarvis,
respond naturally as Jarvis.
"""

        return f"""
{system_prompt}

Conversation Summary:
{self.summary}

Recent Conversation:
{recent_context}

User: {user_input}
Jarvis:
"""

    def chat(self, user_input: str) -> str:
        """
        Main conversation pipeline.
        """

        intent = self.router.classify(user_input)

        self.logger.info(f"Intent: {intent}")
        self.logger.info(f"User input: {user_input}")

        if intent == "tool":
            response = calculate(user_input)

        else:
            prompt = self.build_prompt(user_input)

            try:
                response = self.llm.generate(prompt)

            except Exception as e:
                self.logger.error(f"LLM Error: {e}")
                response = "Sorry, I couldn't process that request."

        self.history.append(("User", user_input))
        self.history.append(("Jarvis", response))

        return response

from pathlib import Path
import time

from faster_whisper import WhisperModel

from backend.core.logger import get_logger


class WhisperSTT:
    """
    Speech-to-Text using Faster-Whisper.
    """

    def __init__(
        self,
        model_size: str = "base",
        device: str = "cpu",
        compute_type: str = "int8",
    ):
        self.logger = get_logger("Whisper")

        self.logger.info("Loading Whisper model...")

        start = time.time()

        self.model = WhisperModel(
            model_size,
            device=device,
            compute_type=compute_type,
        )

        elapsed = round(time.time() - start, 2)

        self.logger.info(
            f"Whisper model loaded in {elapsed}s"
        )

    def transcribe(self, audio_path: str | Path) -> str:
        """
        Transcribe a WAV audio file.

        Parameters
        ----------
        audio_path : str | Path
            Path to the audio file.

        Returns
        -------
        str
            Recognized speech.
        """

        audio_path = Path(audio_path)

        if not audio_path.exists():
            raise FileNotFoundError(
                f"Audio file not found: {audio_path}"
            )

        self.logger.info(
            f"Transcribing {audio_path.name}"
        )

        start = time.time()

        segments, info = self.model.transcribe(
            str(audio_path),
            beam_size=5,
        )

        text = " ".join(
            segment.text.strip()
            for segment in segments
        ).strip()

        elapsed = round(time.time() - start, 2)

        self.logger.info(
            f"Detected language: {info.language}"
        )

        self.logger.info(
            f"Transcription completed in {elapsed}s"
        )

        return text

from pathlib import Path
from datetime import datetime

import numpy as np
import sounddevice as sd
import soundfile as sf

from backend.audio.vad import VoiceActivityDetector
from backend.core.logger import get_logger


class StreamRecorder:
    """
    Continuously listens to the microphone.

    Starts recording when speech is detected.

    Stops recording after prolonged silence.
    """

    def __init__(
        self,
        sample_rate: int = 16000,
        channels: int = 1,
        chunk_ms: int = 30,
    ):

        self.sample_rate = sample_rate
        self.channels = channels
        self.chunk_ms = chunk_ms

        self.chunk_size = int(
            sample_rate * chunk_ms / 1000
        )

        self.logger = get_logger("StreamRecorder")

        self.vad = VoiceActivityDetector()

        self.output_dir = (
            Path(__file__).resolve().parent.parent.parent
            / "recordings"
        )

        self.output_dir.mkdir(exist_ok=True)
    def generate_filename(self) -> Path:
         timestamp = datetime.now().strftime(
             "%Y%m%d_%H%M%S"
         )

         return self.output_dir / f"{timestamp}.wav"
    def record(self) -> Path | None:

        self.logger.info("Listening...")

        buffer = []

        with sd.InputStream(
            samplerate=self.sample_rate,
            channels=self.channels,
            dtype="float32",
            blocksize=self.chunk_size,
        ) as stream:

            while True:

                chunk, _ = stream.read(self.chunk_size)

                chunk = chunk.flatten()

                state = self.vad.update(chunk)

                if state == "idle":
                    continue

                if state == "start":

                    self.logger.info(
                        "Recording started"
                    )

                    buffer = [chunk]

                    continue

                if state == "recording":

                    buffer.append(chunk)

                    continue

                if state == "stop":

                    self.logger.info(
                        "Recording stopped"
                    )

                    audio = np.concatenate(buffer)

                    filename = self.generate_filename()

                    sf.write(
                        filename,
                        audio,
                        self.sample_rate,
                    )

                    self.logger.info(
                        f"Saved {filename.name}"
                    )

                    return filename
